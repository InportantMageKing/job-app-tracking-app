# Frontend Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the React single-page app for the job application tracker. It covers login/signup, the dashboard, the applications list, application detail (with timeline and notes), and resume versions, all talking to the existing Django API through a Vite dev proxy.

**Architecture:** React 19 + TypeScript, built by Vite. All HTTP goes through one small client (`src/api/client.ts`) that adds the token and turns API errors into `ApiError`. Server state lives in TanStack Query hooks, one file per API area. Mutations invalidate the queries they affect. Pages compose small, presentational components, and those components are the ones with unit tests. The app runs in a `node:22-alpine` container under Docker Compose, next to the backend.

**Tech Stack:** React 19, TypeScript 5.8, Vite 7, Tailwind CSS 4 (`@tailwindcss/vite`), TanStack Query 5, React Router 7, Vitest 3 + React Testing Library + jsdom.

**Spec:** `docs/superpowers/specs/2026-10-27-job-app-tracker-design.md`
**Depends on:** the backend plan (`docs/superpowers/plans/2026-10-27-backend-api.md`), which must be complete. Every API endpoint used below already exists and is tested.

### How to run commands (read this first)

- The Bash tool runs **Windows Git Bash**, but the repo, Docker, and git SSH access all live in **WSL**. Run **every** shell command through WSL from the repo root:

  ```bash
  wsl.exe -e bash -ic "cd ~/projects/job-app-tracking-app && <command>"
  ```

  Keep the `-i` flag. It loads `~/.bashrc`, which sets up the SSH agent that git needs. If you see `bash: cannot set terminal process group` or `no job control in this shell`, ignore it; it's harmless noise.
- Use these paths with the Read/Write/Edit tools: `//wsl.localhost/Ubuntu/home/atlas/projects/job-app-tracking-app/<path>`.
- Node isn't installed in WSL, and that's on purpose. **Never** run `npm`, `npx`, or `node` directly; they always run inside the frontend container:

- The container runs as uid/gid `1000:1000`, so `node_modules/` and `package-lock.json` are owned by the WSL user.

### API contract (from the backend; don't change it)

| Endpoint | Notes |
|---|---|
| `POST /api/auth/signup/`, `POST /api/auth/login/` | body `{username, password}` → `{token, username}` |
| `POST /api/auth/logout/` | 204 |
| `GET /api/applications/?status=&source=&search=` | list of `Application` |
| `POST /api/applications/`, `PATCH /api/applications/{id}/`, `DELETE /api/applications/{id}/` | `current_status` is read-only |
| `GET /api/applications/{id}/` | `ApplicationDetail` (has `events`, newest first) |
| `POST /api/applications/{id}/events/` | `{event_type:"note", note}` or `{event_type:"status_change", to_status, note?}` |
| `GET /api/resume-versions/`, `POST` (multipart `name`, `description`, `file`), `DELETE /api/resume-versions/{id}/` | |
| `GET /api/resume-versions/{id}/file/` | file download; needs the token header |
| `GET /api/dashboard/stats/` | `DashboardStats` |
| Errors | DRF JSON, either `{"field": ["msg"]}` or `{"detail": "msg"}`. 401 means the token is invalid. |

### Fixed values

| Thing | Value |
|---|---|
| Status order + labels | `applied` Applied, `phone_screen` Phone Screen, `interview` Interview, `offer` Offer, `rejected` Rejected, `withdrawn` Withdrawn |
| Source labels | `linkedin` LinkedIn, `indeed` Indeed, `company_site` Company Site, `referral` Referral, `other` Other |
| localStorage keys | `jobtracker.token`, `jobtracker.username` |
| Resume files | `.pdf`, `.doc`, `.docx`; max 5 MB (`5 * 1024 * 1024` bytes) |
| Dev URL | http://localhost:5173 (Vite proxies `/api` to the `backend` service) |

```ts
export interface ApplicationEvent {
  id: number;
  event_type: EventType;
  from_status: Status | null;
  to_status: Status | null;
  note: string;
  created_at: string;
}

export interface Application {
  id: number;
  company: string;
  job_title: string;
  source: Source;
  link: string;
  date_applied: string;
  location: string;
  salary_range: string;
  contact_person: string;
  resume_version: number | null;
  resume_version_name: string | null;
  current_status: Status;
  created_at: string;
  updated_at: string;
}

export interface ApplicationDetail extends Application {
  events: ApplicationEvent[];
}

export type ApplicationInput = Pick<
  Application,

  | "job_title"
  | "source"
  | "link"
  | "date_applied"
  | "location"
  | "salary_range"
  | "contact_person"
  | "resume_version"
>;

export type NewEvent =
  | { event_type: "note"; note: string }
```

```ts
import type { Source, Status } from "../api/types";

export const STATUSES: Status[] = ["applied", "phone_screen", "interview", "offer", "rejected", "withdrawn"];

export const STATUS_LABELS: Record<Status, string> = {
  applied: "Applied",
  phone_screen: "Phone Screen",
  interview: "Interview",
  offer: "Offer",
  rejected: "Rejected",
  withdrawn: "Withdrawn",
};

export const STATUS_COLORS: Record<Status, string> = {
  applied: "bg-slate-100 text-slate-700",
  phone_screen: "bg-sky-100 text-sky-800",
  interview: "bg-indigo-100 text-indigo-800",
  offer: "bg-emerald-100 text-emerald-800",
  rejected: "bg-rose-100 text-rose-800",
  withdrawn: "bg-amber-100 text-amber-800",
};

export const SOURCES: Source[] = ["linkedin", "indeed", "company_site", "referral", "other"];

export const SOURCE_LABELS: Record<Source, string> = {
  linkedin: "LinkedIn",
  indeed: "Indeed",
  company_site: "Company Site",
  referral: "Referral",
  other: "Other",
};

/** Parse "YYYY-MM-DD" as a local calendar day (new Date("YYYY-MM-DD") would use UTC midnight). */
function parseDay(day: string): Date {
  const [year, month, date] = day.split("-").map(Number);
  return new Date(year, month - 1, date);
}

export function formatDay(day: string): string {
  return parseDay(day).toLocaleDateString("en-US", { month: "short", day: "numeric", year: "numeric" });
}

export function formatDayShort(day: string): string {
  return parseDay(day).toLocaleDateString("en-US", { month: "short", day: "numeric" });
}

export function formatDateTime(iso: string): string {
  return new Date(iso).toLocaleString("en-US", {
    month: "short",
    day: "numeric",
    year: "numeric",
    hour: "numeric",
    minute: "2-digit",
  });
}

/** Today's date as "YYYY-MM-DD" in the user's time zone. */
export function todayLocal(): string {
  return new Date().toLocaleDateString("en-CA");
}
```

```ts
/** The single place the app talks HTTP: adds the token, prefixes /api, and normalizes errors. */

const TOKEN_KEY = "jobtracker.token";
const USERNAME_KEY = "jobtracker.username";

/** Flatten DRF error bodies ({"field": ["msg"]} or {"detail": "msg"}) into one readable line. */
export function errorMessage(body: unknown): string | null {
  if (!body || typeof body !== "object") return null;
  const parts = Object.entries(body as Record<string, unknown>).map(([key, value]) => {
    const text = Array.isArray(value) ? value.join(" ") : String(value);
    return key === "detail" || key === "non_field_errors" ? text : `${key}: ${text}`;
  });
  return parts.length > 0 ? parts.join(" ") : null;
}

export class ApiError extends Error {
  status: number;
  body: unknown;

  constructor(status: number, body: unknown) {
    super(errorMessage(body) ?? `Request failed (${status}).`);
    this.name = "ApiError";
    this.status = status;
    this.body = body;
  }
}

export const tokenStore = {
  get: () => localStorage.getItem(TOKEN_KEY),
  getUsername: () => localStorage.getItem(USERNAME_KEY),
  set: (token: string, username: string) => {
    localStorage.setItem(TOKEN_KEY, token);
    localStorage.setItem(USERNAME_KEY, username);
  },
  clear: () => {
    localStorage.removeItem(TOKEN_KEY);
    localStorage.removeItem(USERNAME_KEY);
  },
};

let onUnauthorized: () => void = () => {};

/** Register what happens when the API rejects the stored token (AuthProvider logs the user out). */
export function setUnauthorizedHandler(handler: () => void) {
  onUnauthorized = handler;
}

async function send(path: string, init: RequestInit = {}): Promise<Response> {
  const headers = new Headers(init.headers);
  const token = tokenStore.get();
  if (token) headers.set("Authorization", `Token ${token}`);
  // FormData bodies must keep the browser-generated multipart Content-Type.
  if (init.body && !(init.body instanceof FormData) && !headers.has("Content-Type")) {
    headers.set("Content-Type", "application/json");
  }

  const response = await fetch(`/api${path}`, { ...init, headers });

  if (response.status === 401 && token) {
    tokenStore.clear();
    onUnauthorized();
  }
  if (!response.ok) {
    const body = await response.json().catch(() => null);
    throw new ApiError(response.status, body);
  }
  return response;
}

export async function apiFetch<T>(path: string, init?: RequestInit): Promise<T> {
  const response = await send(path, init);
  if (response.status === 204) return undefined as T;
  return (await response.json()) as T;
}

/** Download a protected file. A plain link can't send the token header, so fetch it as a blob. */
export async function apiDownload(path: string, filename: string): Promise<void> {
  const response = await send(path);
  const url = URL.createObjectURL(await response.blob());
  const link = document.createElement("a");
  link.href = url;
  link.download = filename;
  link.click();
  URL.revokeObjectURL(url);
}

export function jsonBody(data: unknown): string {
  return JSON.stringify(data);
}
```

```tsx
import { useQueryClient } from "@tanstack/react-query";
import { createContext, useCallback, useContext, useEffect, useMemo, useState, type ReactNode } from "react";
import { useNavigate } from "react-router";

import { apiFetch, jsonBody, setUnauthorizedHandler, tokenStore } from "../api/client";
import type { AuthResponse } from "../api/types";

interface AuthState {
  username: string | null;
  login: (username: string, password: string) => Promise<void>;
  signup: (username: string, password: string) => Promise<void>;
  logout: () => Promise<void>;
}

const AuthContext = createContext<AuthState | null>(null);

export function AuthProvider({ children }: { children: ReactNode }) {
  const [username, setUsername] = useState<string | null>(() => tokenStore.getUsername());
  const queryClient = useQueryClient();
  const navigate = useNavigate();

  const endSession = useCallback(() => {
    tokenStore.clear();
    setUsername(null);
    queryClient.clear();
    navigate("/login", { replace: true });
  }, [navigate, queryClient]);

  useEffect(() => {
    setUnauthorizedHandler(endSession);
  }, [endSession]);

  const authenticate = useCallback(async (path: string, name: string, password: string) => {
    const data = await apiFetch<AuthResponse>(path, {
      method: "POST",
      body: jsonBody({ username: name, password }),
    });
    tokenStore.set(data.token, data.username);
    setUsername(data.username);
  }, []);

  const value = useMemo<AuthState>(
    () => ({
      username,
      login: (name, password) => authenticate("/auth/login/", name, password),
      signup: (name, password) => authenticate("/auth/signup/", name, password),
      logout: async () => {
        try {
          await apiFetch<void>("/auth/logout/", { method: "POST" });
        } catch {
          // The token may already be invalid; logging out locally is what matters.
        }
        endSession();
      },
    }),
    [username, authenticate, endSession],
  );

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

export function useAuth(): AuthState {
  const context = useContext(AuthContext);
  if (!context) throw new Error("useAuth must be used inside <AuthProvider>");
  return context;
}
```

```ts
import { keepPreviousData, useQuery } from "@tanstack/react-query";

import { apiFetch } from "./client";
import type { Application, Source, Status } from "./types";

export interface ApplicationFilters {
  status?: Status | "";
  source?: Source | "";
  search?: string;
}

export function applicationsPath(filters: ApplicationFilters): string {
  const params = new URLSearchParams();
  for (const [key, value] of Object.entries(filters)) {
    if (value) params.set(key, value);
  }
  const query = params.toString();
  return query ? `/applications/?${query}` : "/applications/";
}

export function useApplications(filters: ApplicationFilters) {
  return useQuery({
    queryKey: ["applications", filters],
    queryFn: () => apiFetch<Application[]>(applicationsPath(filters)),
    // Keep the old rows on screen while a new filter loads, instead of flashing "Loading".
    placeholderData: keepPreviousData,
  });
}
```

```ts
export const RESUME_EXTENSIONS = [".pdf", ".doc", ".docx"];
export const RESUME_MAX_BYTES = 5 * 1024 * 1024;

/** Instant client-side check that mirrors the API's upload rules. The API stays authoritative. */
export function validateResumeFile(file: File | null): string | null {
  if (!file) return "Choose a file to upload.";
  const dot = file.name.lastIndexOf(".");
  const extension = dot === -1 ? "" : file.name.slice(dot).toLowerCase();
  if (!RESUME_EXTENSIONS.includes(extension)) return "Upload a PDF, DOC, or DOCX file.";
  if (file.size > RESUME_MAX_BYTES) return "File must be 5 MB or smaller.";
  return null;
}
```

```tsx
import type { ApplicationEvent } from "../api/types";
import { formatDateTime, STATUS_LABELS } from "../lib/labels";

function describe(event: ApplicationEvent): string {
  if (event.event_type === "note") return "Note";
  if (event.from_status === null) return "Application created";
  return `${STATUS_LABELS[event.from_status]} → ${event.to_status ? STATUS_LABELS[event.to_status] : "?"}`;
}

export function Timeline({ events }: { events: ApplicationEvent[] }) {
  if (events.length === 0) return <p className="text-sm text-slate-500">No activity yet.</p>;

  return (
    <ol className="space-y-3">
      {events.map((event) => (
        <li
          key={event.id}
          className={`rounded-md border p-3 ${
            event.event_type === "note" ? "border-slate-200 bg-slate-50" : "border-slate-300 bg-white"
          }`}
        >
          <div className="flex flex-wrap items-center justify-between gap-2 text-sm">
            <span className="font-medium">{describe(event)}</span>
            <time className="text-xs text-slate-500" dateTime={event.created_at}>
              {formatDateTime(event.created_at)}
            </time>
          </div>
          {event.note && <p className="mt-1 whitespace-pre-wrap text-sm text-slate-700">{event.note}</p>}
        </li>
      ))}
    </ol>
  );
}
```

```tsx
import { useState, type FormEvent } from "react";

import type { NewEvent, Status } from "../api/types";
import { STATUS_LABELS, STATUSES } from "../lib/labels";
import { ui } from "./ui";

interface Props {
  currentStatus: Status;
  onSubmit: (event: NewEvent) => Promise<unknown>;
}

type Mode = "note" | "status_change";

export function AddEventForm({ currentStatus, onSubmit }: Props) {
  const [mode, setMode] = useState<Mode>("note");
  const [toStatus, setToStatus] = useState<Status | "">("");
  const [note, setNote] = useState("");
  const [error, setError] = useState<string | null>(null);
  const [submitting, setSubmitting] = useState(false);

  function switchMode(next: Mode) {
    setMode(next);
    setError(null);
  }

  async function handleSubmit(event: FormEvent) {
    event.preventDefault();
    const text = note.trim();
    let payload: NewEvent;
    if (mode === "note") {
      if (!text) {
        setError("Write a note before saving.");
        return;
      }
      payload = { event_type: "note", note: text };
    } else {
      if (!toStatus) {
        setError("Pick the new status.");
        return;
      }
      payload = { event_type: "status_change", to_status: toStatus, note: text };
    }

    setError(null);
    setSubmitting(true);
    try {
      await onSubmit(payload);
      setNote("");
      setToStatus("");
    } catch (caught) {
      setError(caught instanceof Error ? caught.message : "Something went wrong.");
    } finally {
      setSubmitting(false);
    }
  }

  const modeButton = (value: Mode, label: string) => (
    <button
      type="button"
      aria-pressed={mode === value}
      onClick={() => switchMode(value)}
      className={`flex-1 rounded-md px-3 py-1.5 text-sm font-medium ${
        mode === value ? "bg-slate-900 text-white" : "text-slate-600 hover:bg-slate-100"
      }`}
    >
      {label}
    </button>
  );

  return (
    <form onSubmit={handleSubmit} className="space-y-3">
      <div className="flex gap-1 rounded-lg border border-slate-200 p-1">
        {modeButton("note", "Add note")}
        {modeButton("status_change", "Change status")}
      </div>

      {mode === "status_change" && (
        <label className={ui.label}>
          New status
          <select
            className={ui.input}
            value={toStatus}
            onChange={(event) => setToStatus(event.target.value as Status | "")}
          >
            <option value="">Choose…</option>
            {STATUSES.filter((status) => status !== currentStatus).map((status) => (
              <option key={status} value={status}>
                {STATUS_LABELS[status]}
              </option>
            ))}
          </select>
        </label>
      )}

      <label className={ui.label}>
        {mode === "note" ? "Note" : "Note (optional)"}
        <textarea
          className={ui.input}
          rows={3}
          placeholder={mode === "note" ? "e.g. Recruiter emailed to schedule a call" : "Anything worth remembering"}
          value={note}
          onChange={(event) => setNote(event.target.value)}
        />
      </label>

      {error && (
        <p role="alert" className={ui.error}>
          {error}
        </p>
      )}
      <button type="submit" className={ui.primary} disabled={submitting}>
        {submitting ? "Saving…" : "Save"}
      </button>
    </form>
  );
}
```

```ts
export function useApplication(id: number) {
  return useQuery({
    queryKey: ["application", id],
    queryFn: () => apiFetch<ApplicationDetail>(`/applications/${id}/`),
  });
}

export function useAddEvent(id: number) {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: (event: NewEvent) =>
      apiFetch<ApplicationEvent>(`/applications/${id}/events/`, { method: "POST", body: jsonBody(event) }),
    onSuccess: () => {
      void queryClient.invalidateQueries({ queryKey: ["application", id] });
      void queryClient.invalidateQueries({ queryKey: ["applications"] });
      void queryClient.invalidateQueries({ queryKey: ["dashboard"] });
    },
  });
}

export function useDeleteApplication() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: (id: number) => apiFetch<void>(`/applications/${id}/`, { method: "DELETE" }),
    onSuccess: (_result, id) => {
      queryClient.removeQueries({ queryKey: ["application", id] });
      void queryClient.invalidateQueries({ queryKey: ["applications"] });
      void queryClient.invalidateQueries({ queryKey: ["dashboard"] });
      void queryClient.invalidateQueries({ queryKey: ["resume-versions"] });
    },
  });
}
```

```tsx
import { useState } from "react";
import { Link, useNavigate, useParams } from "react-router";

import { toApplicationInput, useAddEvent, useApplication, useDeleteApplication, useUpdateApplication } from "../api/applications";
import { ApiError } from "../api/client";
import { AddEventForm } from "../components/AddEventForm";
import { ApplicationDetails } from "../components/ApplicationDetails";
import { ApplicationForm } from "../components/ApplicationForm";
import { StatusBadge } from "../components/StatusBadge";
import { Timeline } from "../components/Timeline";
import { ui } from "../components/ui";

export function ApplicationDetailPage() {
  const id = Number(useParams().id);
  const navigate = useNavigate();
  const application = useApplication(id);
  const update = useUpdateApplication(id);
  const addEvent = useAddEvent(id);
  const remove = useDeleteApplication();
  const [editing, setEditing] = useState(false);
  const [confirmingDelete, setConfirmingDelete] = useState(false);
  const [deleteError, setDeleteError] = useState<string | null>(null);

  async function handleDelete() {
    setDeleteError(null);
    try {
      await remove.mutateAsync(id);
      navigate("/applications", { replace: true });
    } catch (caught) {
      setDeleteError(caught instanceof Error ? caught.message : "Delete failed.");
    }
  }

  if (application.isPending) return <p className="text-slate-500">Loading…</p>;
  if (application.isError) {
    const notFound = application.error instanceof ApiError && application.error.status === 404;
    return (
      <div className="space-y-2">
        <p role="alert" className={ui.error}>
          {notFound ? "This application doesn't exist." : application.error.message}
        </p>
        <Link to="/applications" className="text-sm text-slate-600 hover:underline">
          ← Back to applications
        </Link>
      </div>
    );
  }

  const data = application.data;

  return (
    <div className="space-y-6">
      <Link to="/applications" className="text-sm text-slate-500 hover:underline">
        ← All applications
      </Link>

      <div className="flex flex-wrap items-start justify-between gap-3">

          <h1 className={ui.heading}>{data.company}</h1>

      </div>

      <div className="grid gap-6 lg:grid-cols-3">
        <section className={`${ui.card} lg:col-span-2`}>
          {editing ? (
            <ApplicationForm
              initial={toApplicationInput(data)}
              submitLabel="Save changes"
              onSubmit={async (input) => {
                await update.mutateAsync(input);
                setEditing(false);
              }}
              onCancel={() => setEditing(false)}
            />
          ) : (
            <ApplicationDetails application={data} onEdit={() => setEditing(true)} />
          )}
        </section>

        <section className={`${ui.card} space-y-4`}>
          <h2 className="font-semibold">Activity</h2>
          <AddEventForm currentStatus={data.current_status} onSubmit={(event) => addEvent.mutateAsync(event)} />
          <Timeline events={data.events} />
        </section>
      </div>

      <section className="flex flex-wrap items-center gap-2 border-t border-slate-200 pt-4">
        {confirmingDelete ? (
          <>
            <span className="text-sm text-slate-600">Delete this application and its whole timeline?</span>
            <button type="button" className={ui.danger} disabled={remove.isPending} onClick={() => void handleDelete()}>
              Delete
            </button>
            <button type="button" className={ui.secondary} onClick={() => setConfirmingDelete(false)}>
              Cancel
            </button>
          </>
        ) : (
          <button type="button" className={ui.secondary} onClick={() => setConfirmingDelete(true)}>

          </button>
        )}
        {deleteError && (
```

```tsx
import type { WeekStat } from "../../api/types";
import { formatDayShort } from "../../lib/labels";

export function WeeklyChart({ weeks }: { weeks: WeekStat[] }) {
  if (weeks.length === 0) return null;
  const busiest = Math.max(1, ...weeks.map((week) => week.count));

  return (
    <div>
      <ul className="flex h-40 items-end gap-1.5">
        {weeks.map((week) => {
          const label = `Week of ${formatDayShort(week.week_start)}: ${week.count} application${week.count === 1 ? "" : "s"}`;
          return (
            <li key={week.week_start} aria-label={label} title={label} className="flex h-full flex-1 flex-col justify-end">
              <div className="rounded-t bg-slate-800" style={{ height: `${(week.count / busiest) * 100}%` }} />
            </li>
          );
        })}
      </ul>
      <div className="mt-2 flex justify-between text-xs text-slate-500">
        <span>{formatDayShort(weeks[0].week_start)}</span>
        <span>{formatDayShort(weeks[weeks.length - 1].week_start)}</span>
      </div>
    </div>
  );
}
```
